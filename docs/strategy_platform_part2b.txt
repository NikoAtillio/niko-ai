SECTION 11 — PLATFORM DETECTOR PSEUDOCODE (TypeScript-style)

11.1  Core Data Types
interface Candle {   timestamp: Date;   open: number; high: number; low: number; close: number;   volume: number; }  interface Signal {   strategy_id: string;   direction: 'long' | 'short';   regime: RegimeType;   entry_price: number;   stop_loss: number;   take_profit_1: number;   take_profit_2: number;   confidence: number;           // 0.0 – 1.0   rules_met: Record<string, boolean>;   instrument: string;   entry_timeframe: string;   signal_time: Date; }  type RegimeType = 'STRONG_TREND' | 'WEAK_TREND' | 'RANGE' | 'HIGH_VOL';  const REGIME_ACTIVATION_MAP: Record<RegimeType, string[]> = {   STRONG_TREND: ['TREND_PULLBACK','MOMENTUM_CONT','BREAKOUT_CONFIRM','ORB'],   WEAK_TREND:   ['TREND_PULLBACK','BREAKOUT_CONFIRM','MEAN_REVERT_BAND','ORB','VOL_COMPRESS_BREAK'],   RANGE:        ['RANGE_REVERSION','MEAN_REVERT_BAND','VOL_COMPRESS_BREAK','ORB'],   HIGH_VOL:     ['BREAKOUT_CONFIRM','ORB'], };  const STRATEGY_PRIORITY: string[] = [   'ORB','TREND_PULLBACK','MOMENTUM_CONT','BREAKOUT_CONFIRM',   'VOL_COMPRESS_BREAK','RANGE_REVERSION','MEAN_REVERT_BAND' ];
11.2  Universal Candle Utilities
const candleBody  = (c: Candle) => Math.abs(c.close - c.open); const candleRange = (c: Candle) => c.high - c.low; const bodyPct     = (c: Candle) => candleBody(c) / candleRange(c);  function wickRatio(c: Candle, minBodyPct = 0.10): number | null {   if (bodyPct(c) < minBodyPct) return null;  // doji guard   const upper = c.high - Math.max(c.open, c.close);   const lower = Math.min(c.open, c.close) - c.low;   return Math.max(upper, lower) / candleBody(c); }  function isRejectionBullish(c: Candle, wickMin = 1.5, minBodyPct = 0.10): boolean {   if (bodyPct(c) < minBodyPct) return false;   const lower = Math.min(c.open, c.close) - c.low;   const upper = c.high - Math.max(c.open, c.close);   return c.close > c.open && (lower / candleBody(c)) >= wickMin && lower > upper; }  function isRejectionBearish(c: Candle, wickMin = 1.5, minBodyPct = 0.10): boolean {   if (bodyPct(c) < minBodyPct) return false;   const upper = c.high - Math.max(c.open, c.close);   const lower = Math.min(c.open, c.close) - c.low;   return c.close < c.open && (upper / candleBody(c)) >= wickMin && upper > lower; }
11.3  Confidence Scorer
const MANDATORY = 3, STRONG = 2, OPTIONAL = 1;  function scoreSignal(   rules: Array<{ name: string; weight: number; met: boolean }> ): number {   // Hard block: any mandatory rule fails → 0.0   for (const r of rules)     if (r.weight === MANDATORY && !r.met) return 0.0;    const total  = rules.reduce((s, r) => s + r.weight, 0);   const earned = rules.filter(r => r.met).reduce((s, r) => s + r.weight, 0);   return earned / total; }  // Example: TREND_PULLBACK — 8 rules met out of 9, volume failed // Rules: 3×MANDATORY(met) + 1×STRONG(failed) + 1×STRONG(met) + 3×OPTIONAL(met) // earned = 9+2+3 = 14, total = 9+2+2+3 = 16 → confidence = 0.875
11.4  Regime Classifier with Hysteresis
function classifyRegime(   candles: Candle[], prevRegime: RegimeType, p: StrategyParams ): RegimeType {   if (candles.length < p.min_bars_history) return prevRegime;    const adx      = calcADX(candles, p.adx_period).at(-1)!;   const ema50    = calcEMA(candles, 50);   const slope    = (ema50.at(-1)! - ema50.at(-5)!) / ema50.at(-5)!;   const atrNow   = calcATR(candles, p.atr_period).at(-1)!;   const atrAvg   = mean(calcATR(candles, p.atr_period).slice(-20));   const atrRatio = atrNow / atrAvg;   const bbWidth  = calcBBWidth(candles, 20, 2.0).at(-1)!;    // HIGH_VOL — highest priority   if (prevRegime === 'HIGH_VOL' && atrRatio >= (p.atr_high_vol_exit ?? 1.5)) return 'HIGH_VOL';   if (atrRatio > (p.atr_high_vol_enter ?? 2.0)) return 'HIGH_VOL';    // STRONG_TREND with hysteresis   const inStrong = prevRegime === 'STRONG_TREND';   if (inStrong && adx >= (p.adx_strong_trend_exit ?? 22)) return 'STRONG_TREND';   if (!inStrong && adx > (p.adx_strong_trend_enter ?? 25)       && Math.abs(slope) > (p.ema_slope_threshold ?? 0.001)) return 'STRONG_TREND';    // RANGE with hysteresis   const inRange = prevRegime === 'RANGE';   if (inRange && adx <= (p.adx_range_exit ?? 21)) return 'RANGE';   if (!inRange && adx < (p.adx_range_enter ?? 18)       && bbWidth < (p.bb_width_range_threshold ?? 0.03)) return 'RANGE';    return 'WEAK_TREND'; }
11.5  Main Platform Loop (per bar)
function onNewBar(   candles: Candle[], params: StrategyParams, state: PlatformState ): Signal[] {    // Guard: minimum history   if (candles.length < params.min_bars_history) return [];    // Step 1: Classify regime with hold-bar hysteresis   const candidate = classifyRegime(candles, state.regime, params);   if (candidate !== state.regime) {     state.regimeBarCount = (state.regimeBarCount ?? 0) + 1;     if (state.regimeBarCount >= (params.regime_hold_min_bars ?? 3)) {       state.regime = candidate;       state.regimeBarCount = 0;     }   } else { state.regimeBarCount = 0; }    // Step 2: Run allowed strategy detectors   const allowed = REGIME_ACTIVATION_MAP[state.regime];   const signals: Signal[] = [];    const detectors: Record<string, () => Signal[]> = {     TREND_PULLBACK:    () => detectTrendPullback(candles, state.regime, params),     BREAKOUT_CONFIRM:  () => detectBreakout(candles, state.regime, params),     RANGE_REVERSION:   () => detectRangeReversion(candles, state.regime, params),     MEAN_REVERT_BAND:  () => detectMeanReversion(candles, state.regime, params),     VOL_COMPRESS_BREAK:() => detectVolCompression(candles, state.regime, params),     ORB:               () => detectORB(candles, state.regime, params, state.session),     MOMENTUM_CONT:     () => detectMomentumCont(candles, state.regime, params),   };    for (const id of allowed) {     if (params.active_strategies.includes(id))       signals.push(...detectors[id]());   }    // Step 3: Resolve conflicts and return   return resolveAllSignals(signals, state.regime, state.openPositions, params); }
11.6  Indicator Caching Strategy
All indicators are pre-computed once per bar and cached. Strategies read from cache — never recompute. This ensures consistent values across all detectors and avoids performance issues.
interface IndicatorCache {   ema20: number[]; ema50: number[]; ema200: number[];   atr14: number[]; adx14: number[];   bb_upper: number[]; bb_lower: number[]; bb_mid: number[]; bb_std: number[];   kc_upper: number[]; kc_lower: number[];   vwap: number[];    // intraday reset at session open   volume_avg20: number[];   bb_width: number[]; }  function buildCache(candles: Candle[], params: StrategyParams): IndicatorCache {   return {     ema20:       calcEMA(candles, 20),     ema50:       calcEMA(candles, 50),     ema200:      calcEMA(candles, 200),     atr14:       calcATR(candles, params.atr_period),     adx14:       calcADX(candles, params.adx_period),     bb_upper:    calcBBUpper(candles, 20, params.bb_std),     bb_lower:    calcBBLower(candles, 20, params.bb_std),     bb_mid:      calcEMA(candles, 20),  // same as ema20     bb_std:      calcBBStd(candles, 20),     kc_upper:    calcKCUpper(candles, 20, params.kc_mult ?? 1.5),     kc_lower:    calcKCLower(candles, 20, params.kc_mult ?? 1.5),     vwap:        calcVWAP(candles),     volume_avg20:calcRollingMean(candles.map(c => c.volume), 20),     bb_width:    calcBBWidth(candles, 20, params.bb_std),   }; }

12.2  Universal Rules — Always Apply
Minimum 200 bars of history required before ANY signal is generated (hard minimum)
Wick ratio formula: longer_wick / candle_body (NOT range-relative)
Wick ratio threshold: >= 1.5 across all strategies (configurable per strategy)
Rejection candle: bullish = close > open + lower wick dominant; bearish = inverse
Confidence = Σ(weight × met) / Σ(weight). Any MANDATORY rule failure → confidence = 0.0
Regime runs FIRST on every bar before any strategy detector
Regime hysteresis: enter/exit thresholds are asymmetric (3-point ADX band)
Indicator periods are bar-count based — same on M1, M5, H1, D1
EMA used throughout (not SMA) — more responsive to recent price action
Volume filter: tick volume for Forex/Metals, real volume for Crypto/Equities
Opposing signals within 0.10 confidence gap → NO TRADE
All times in UTC. DST handled by converting exchange local time at runtime
